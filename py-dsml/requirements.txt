# Figure measurement -- reading back the PNGs that rsvg-convert produces. pillow
# already arrives as a matplotlib dependency, so this install is a no-op; it is
# declared because figure scripts import it directly, and a direct import should
# not rest on another package's dependency tree.
pillow

# Perceptual image hashing. Backs the image-dedup leakage screen: near-duplicate
# images split across train and test inflate scores without any model learning
# anything, and exact-hash comparison misses them because a resize or
# requantization changes every byte.
imagehash

# Google Drive fetches from inside notebooks. Every lecture notebook nbgen emits
# carries a `display_image` helper that imports gdown at cell top to pull figures
# and animations by Drive share id. Colab preinstalls it, so the gap only shows
# up in headless execution: without it, the import fails before any figure cell
# runs and every lecture master fails the execution gate on a helper it never
# calls.
gdown
